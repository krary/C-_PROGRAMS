static gboolean reprotext(gpointer objeto){
	
	DatosEscritura *datos = (DatosEscritura*)objeto;
	if(datos->texto[datos->indice] != '\0'){
		GtkTextIter iter;
		gtk_text_buffer_get_end_iter(datos->buffer,&iter);
		char letra[2]={datos->texto[datos->indice],'\0'};
		gtk_text_buffer_insert(datos->buffer,&iter,letra,-1);
		datos->indice++;
		return G_SOURCE_CONTINUE; 
		
		}
	g_free(datos);
	G_SOURCE_REMOVE;

typedef struct {
     int start_x,end_x,start_y,end_y;

}ConfigDrawing;

static void dibujar(Gtk_DrawingArea *drawing,cairo_t *cr,int width,int height,gpointer ptr){

ConfigDrawing *config = (ConfigDrawing)ptr;
cairo_set_source_rgb(cr,0.0,0.0,1.0);
cairo_set_line_width(cr,1.0) //Estableciondo el grosor

cairo_move_to(cr,config->start_x,config->start_y);
cairo_line_to(cr,x,config->start_y - config->end_y); //si queremos movernos hacia abajo

cairo_line_to(cr,config->start-x - config->end_x,config->start_y); //si queremos movernos hacia el lado
cairo_stroke(cr);
}

closedir()  CERRAR DIRECTORIO

static void on_selected_string(GtkDropDown *dropdown,GParamSpec *prspec,gpointer user_data){
	guint selected = gtk_drop_down_get_selected(dropdown);
	if(selected == GTK_INVALID_LIST_POSITION) return;
	GListModel *model = gtk_drop_down_get_model(dropdown);
	GtkStringObject *obj = g_list_model_get_item(model,selected);
	
	if(obj != NULL){
		const char *texto = gtk_string_object_get_string(obj);
		g_object_unref(obj);
		printf("[%s]\n",texto);}
		}	
	DENTRO DE NUESTRO ACTIVATE +
	g_signal_connect(dropdown,"notify::selected",G_CALLBACK(on_selected_string),NULL);
